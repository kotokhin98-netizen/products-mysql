=== ШАГ 1. Создание базы данных ===
1. mkdir products-mysql
2. cd products-mysql
3. cat > my.cnf << 'CNF'
   [mysqld]
   character-set-server=utf8mb4
   collation-server=utf8mb4_unicode_ci
   [client]
   default-character-set=utf8mb4
   [mysql]
   default-character-set=utf8mb4
   CNF
4. docker run --name products-mysql -e MYSQL_ROOT_PASSWORD=123 -e LANG=C.UTF-8 -v mysql_data:/var/lib/mysql -v $(pwd)/my.cnf:/etc/mysql/conf.d/my.cnf:ro -d mysql:8.0
5. docker ps
6. docker exec -it products-mysql mysql --default-character-set=utf8mb4 -uroot -p
7. CREATE DATABASE shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
8. USE shop;
9. SHOW DATABASES;

=== ШАГ 2. Создание таблиц ===
10. CREATE TABLE products1 (name VARCHAR(100) NOT NULL, price INT NOT NULL) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
11. CREATE TABLE products2 (name VARCHAR(100) NOT NULL, price INT NOT NULL) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
12. SHOW TABLES;

=== ШАГ 3. Вставка данных ===
13. INSERT INTO products1 (name, price) VALUES ('Дыни', 800), ('Клубника', 150), ('Яблоки', 120), ('Лимоны', 200);
14. INSERT INTO products2 (name, price) VALUES ('Дыни', 800), ('Клубника', 150), ('Каштаны', 200), ('Хурма', 350);
15. SELECT * FROM products1;
16. SELECT * FROM products2;

=== ШАГ 4. SQL-операции ===
17. SELECT * FROM products1 WHERE price > 150;
18. SELECT name FROM products1;
19. SELECT name FROM products1 UNION SELECT name FROM products2;
20. SELECT name, price FROM products1 EXCEPT SELECT name, price FROM products2;
21. SELECT name, price FROM products1 INTERSECT SELECT name, price FROM products2;
22. SELECT t1.name AS name1, t1.price AS price1, t2.name AS name2, t2.price AS price2 FROM products1 t1 CROSS JOIN products2 t2;
23. SELECT t1.name, t1.price AS price1, t2.price AS price2 FROM products1 t1 JOIN products2 t2 ON t1.name = t2.name;
24. SELECT t1.name, t1.price AS price1, t2.price AS price2 FROM products1 t1 LEFT JOIN products2 t2 ON t1.name = t2.name;

=== ШАГ 5. Создание дампа ===
25. \q
26. docker exec -e MYSQL_PWD=123 products-mysql mysqldump -uroot shop > dump.sql
27. cat dump.sql

=== ШАГ 6. Удаление БД ===
28. docker exec -it products-mysql mysql --default-character-set=utf8mb4 -uroot -p
29. DROP DATABASE shop;
30. SHOW DATABASES;
31. \q

=== ШАГ 7. Восстановление БД из дампа ===
32. docker exec -it products-mysql mysql --default-character-set=utf8mb4 -uroot -p
33. CREATE DATABASE shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
34. \q
35. docker exec -i products-mysql mysql -uroot -p123 --force shop < dump.sql
36. docker exec -it products-mysql mysql --default-character-set=utf8mb4 -uroot -p
37. SHOW DATABASES;
38. USE shop;
39. SHOW TABLES;
40. SELECT * FROM products1;
41. SELECT * FROM products2;
